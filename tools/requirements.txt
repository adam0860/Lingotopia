# Deck generation pipeline. Install with:
#   pip install -r tools/requirements.txt
#   python -m spacy download ru_core_news_sm
#   python -m spacy download es_core_news_sm
#   python -m spacy download ja_core_news_sm

wordfreq>=3.0          # frequency lists
spacy>=3.7             # lemmatisation
jamdict                # JMdict interface (Japanese)
jamdict-data           # the JMdict database itself

# wordfreq needs a Japanese tokeniser; without these, --lang ja fails with
# ModuleNotFoundError: No module named 'MeCab'
mecab-python3
unidic-lite

# Arabic only. CAMeL Tools also needs its databases, which are separate
# downloads — see the Arabic commands in README.md § Generating decks.
# Note CALIMA is GPL v2, unlike the permissive licences above.
camel-tools

# Mandarin Chinese (tools/generate_deck_chinese.py). wordfreq's Chinese
# tokeniser needs jieba too. jieba's sdist fails to build on some Python 3.13
# systems; install setuptools and wheel first (pip install setuptools wheel),
# or use a virtualenv.
jieba>=0.42           # word segmentation (MIT)
pypinyin>=0.51        # pinyin for sentences and tie-breaking (MIT)
opencc-python-reimplemented  # Traditional -> Simplified (Apache-2.0)

# Korean (tools/generate_deck_korean.py). Kiwi bundles its own model.
kiwipiepy>=0.18       # morphological analyser / lemmatiser (LGPL)

# Italian (tools/generate_deck_italian.py): spaCy lemmatiser, plus its model:
#   python -m spacy download it_core_news_sm
# (spacy and wordfreq are already listed above.)

# Portuguese (tools/generate_deck_portuguese.py): spaCy lemmatiser, plus its model:
#   python -m spacy download pt_core_news_sm

# Turkish (tools/generate_deck_turkish.py): Zeyrek is a Python port of Zemberek,
# a rule-based Turkish morphological analyser (spaCy has no Turkish model).
zeyrek>=0.1.3

# Russian reading passages (tools/build_readings_ru.py): pymorphy3 finds each
# word's dictionary form and grammatical form; its Russian dictionaries are a
# separate package.
pymorphy3>=2.0
pymorphy3-dicts-ru

# Arabic reading passages (tools/build_readings_ar.py): camel-tools (above) with
# the two MSA databases, morphology-db-msa-r13 and disambig-mle-calima-msa-r13.
